\section{Datos}
%% ============================================================
%% El problema
\begin{frame}{Ser pobre depende del ingreso, y el ingreso es justo lo que no podemos preguntar}
  \begin{columns}[T]
    \column{0.62\textwidth}
      \small
      Pobre: ingreso por persona bajo la línea; el test no lo trae.
      \begin{itemize}
        \item Preguntarlo es caro y poco confiable.
        \item ``Nadie es pobre'' acierta 80\,\%: usamos F1.
      \end{itemize}
    \column{0.34\textwidth}
      \centering
      \vspace{0.3cm}
      \keyfig{164.960}{hogares}\\[0.4cm]
      \keyfig{1 de 5}{es pobre}
  \end{columns}
  \takeaway{Reconstruiremos el ingreso con lo que sí se ve.}
  \fuente{Fuente: views/tablas/variables\_ausentes\_test\_*.csv; views/tablas/desbalance\_clases.csv.}
  \note{
    \textbf{Qué decir (1 min).} La regla de pobreza es mecánica: si el ingreso por persona
    del hogar está bajo la línea de su ciudad, el hogar es pobre. Tenemos 164.960 hogares
    para entrenar y 66.168 para predecir. El problema es que el test no trae ningún
    ingreso (79 variables solo existen en train, entre ellas todos los ingresos), igual que pasaría en una encuesta corta del gobierno:
    preguntar el ingreso es caro, la gente lo subreporta y además cambia mes a mes.
    Además, solo uno de cada cinco hogares es pobre. Un modelo que diga que nadie es
    pobre acierta en el 80\,\% de los casos, pero no encuentra a ningún pobre. Por eso usamos el
    F1, que castiga tanto dejar pobres por fuera como darle el subsidio a quien no lo
    necesita.\\[3pt]
    \textbf{Pregunta del profesor: ¿por qué no predecir el ingreso y aplicar la línea?}
    \textbf{Respuesta:} El taller pide clasificar, y eso hace el modelo final. El ingreso
    lo usamos solo para enseñarle a un modelo auxiliar, nunca como variable del hogar.
  }
\end{frame}

%% La idea
\begin{frame}{Cuántos trabajan y cuántos dependen ya separan a pobres de no pobres}
  \begin{columns}[c]
    \column{0.62\textwidth}
      \includegraphics[width=\linewidth]{correlatos_pobreza}
    \column{0.36\textwidth}
      \footnotesize
      \textbf{Ingreso por persona} $=$ cuántos trabajan $\times$ cuánto gana cada uno
      $\div$ cuántos comen.
      \begin{itemize}
        \item Más dependientes: pobreza de 8\,\% a 56\,\%.
        \item Jefe sin estudios: 40\,\%; con universidad: 8\,\%.
        \item Sin ocupados (29\,\%) es menos pobre que un tercio ocupado (39\,\%):
          pensiones.
      \end{itemize}
  \end{columns}
  \takeaway{Ninguna variable sola basta: la mejor da un F1 de 0,47.}
  \fuente{Tasa de pobreza en los 164.960 hogares de train; promedio: 20\,\%. Fuente: scripts/13\_figuras\_presentacion.R; views/tablas/leaderboard\_interno.csv.}
  \note{
    \textbf{Qué decir (1 min).} Si no vemos el ingreso, partimos de cómo se forma: cuántas
    personas trabajan, cuánto gana cada una y entre cuántas se reparte. Los datos
    confirman cada pieza. Cuando hay más dependientes por adulto, la pobreza pasa de 8 a
    56\,\%. La educación del jefe también separa mucho. Y la vivienda importa: quien paga
    arriendo es más pobre que quien tiene casa propia pagada. Hay una sorpresa: los
    hogares donde nadie trabaja no son los más pobres, porque muchos son pensionados. Por
    eso la relación no es lineal, y por eso ninguna variable sola basta.\\[3pt]
    \textbf{Pregunta del profesor: ¿basta con estas variables en un Logit?}
    \textbf{Respuesta:} No: un Logit con 5 variables de la teoría da F1 de 0,46. Sube a
    0,72 cuando construimos 48 variables a partir de cada persona del hogar.
  }
\end{frame}

  \end{columns}
  \takeaway{El ingreso del train solo sirve para enseñar; nunca entra como dato del hogar.}
  \fuente{F1 en los datos apartados (holdout). Boosting: LightGBM. Fuente: views/tablas/nivel7/ablacion\_etapa1.csv, resumen\_etapa2.csv, etapa6\_ablacion.csv.}
  \note{
    \textbf{Qué decir (1 min).} Esta es la idea central. En train sí sabemos cuánto gana
    cada persona. Entonces entrenamos un modelo que aprende a predecir ese ingreso usando
    solo lo que el test también trae: si trabaja, cuántas horas, si cotiza a pensión, su
    oficio. En el test aplicamos ese modelo a cada persona, sumamos por hogar, dividimos
    por el número de personas y por la línea. Ese número se vuelve la variable más
    poderosa del clasificador. Solo el ingreso laboral estimado sube el F1 más de un
    punto.\\[3pt]
    \textbf{Pregunta del profesor: ¿eso no es usar la respuesta para predecir?}
    \textbf{Respuesta:} No, porque el modelo que estima el ingreso de un hogar nunca vio
    ese hogar, ni en el entrenamiento ni en la validación. Lo verificamos con pruebas
    automáticas y con una prueba de que train y test se parecen (apéndice).
  }
\end{frame}

\begin{frame}{Saber cuánto gana cada persona vale más que el algoritmo}
  \centering\includegraphics[width=\linewidth]{escalera_completa}
  \takeaway{Variables: $+0{,}29$; algoritmo: $+0{,}03$. Entender el hogar rinde nueve veces más.}
  \fuente{F1 en datos apartados, un paso a la vez. Fuente: views/tablas/leaderboard\_interno.csv y views/tablas/nivel7/.}
  \note{
    \textbf{Qué decir (1 min).} Esta escalera resume todo el proceso. Cada punto es un paso,
    medido siempre en los mismos datos apartados. Los puntos amarillos cambian las
    variables y dejan fijo el algoritmo; los negros cambian el algoritmo y dejan fijas las
    variables. Si sumamos, las variables aportaron 0,29 de F1 y los algoritmos 0,03. El
    salto más grande no tuvo nada de sofisticado: el mismo Logit, pero con variables
    construidas a partir de las personas del hogar.\\[3pt]
    \textbf{Pregunta del profesor: ¿no depende del orden en que hicieron los pasos?}
    \textbf{Respuesta:} Sí, la suma es secuencial. Pero el contraste se sostiene en
    cualquier orden: con LightGBM fijo las variables nuevas suman 0,032, más que toda la
    distancia entre el Logit y el boosting con las mismas variables.
  }
\end{frame}

    \textbf{Pregunta del profesor: esta medida de importancia favorece variables
    continuas.}
    \textbf{Respuesta:} Es cierto, por eso la contrastamos con un boosting interpretable
    (EBM) que mide el aporte de otra forma. Ahí también mandan el ingreso estimado y el

  \raggedright\small

  \takeaway{Repetir todo con otras particiones da el mismo F1: 0,775.}
  \fuente{Fuente: views/tablas/resumen\_particion.csv; views/tablas/nivel7/etapa16\_fase1\_resumen.csv (3 particiones nuevas, 15 evaluaciones).}
  \note{
    \textbf{Qué decir (1 min).} Así entrenamos. Al principio separamos un 20\,\% de los
    hogares y no lo tocamos. Con el otro 80\,\% hacemos validación cruzada: se entrena con
    cuatro bloques y se predice el quinto, rotando. Así, cada hogar recibe una predicción
    de un modelo que no lo vio, y con esas predicciones elegimos todo: hiperparámetros,
    umbral y pesos de la mezcla. ¿Por qué gana el boosting? Porque va sumando árboles
    pequeños que corrigen los errores que quedan, y esos errores están justo en los
    hogares difíciles, cerca de la línea. Al final repetimos todo con tres particiones
    nuevas y el resultado no cambió.\\[3pt]

    sobreajustaron?}

    comparación final la hicimos con particiones nuevas. Además, Kaggle fue la prueba
    externa: 0,78 en la tabla final, igual que en nuestros datos apartados.
  }
\end{frame}

    eligiendo con validación cruzada. La configuración ganadora es prudente: muchos

    pobres deja el F1 igual. Lo único que cambia es el umbral, porque balancear solo
    infla las probabilidades.\\[3pt]
    \textbf{Pregunta del profesor: ¿por qué no afinar con los datos apartados?}
    \textbf{Respuesta:} Porque dejarían de ser una medida limpia. Todo el afinamiento usó
    solo la validación cruzada del 80\,\%.

  \centering\includegraphics[width=\linewidth]{techo_dos_paneles}\par
  \raggedright\footnotesize
  \begin{columns}[T]
    \column{0.58\textwidth} \textbf{A.} Con el ingreso real, el F1 llega a 0,89.
    \column{0.40\textwidth} \textbf{B.} Ahí está el 32\,\% de los pobres.
  \end{columns}
  \takeaway{El límite es la información que falta, no el algoritmo.}
  \fuente{Datos apartados, versión previa del modelo (0,776). Más datos y 16 alternativas: apéndice. Fuente: views/tablas/nivel7/techo\_oraculo.csv, techo\_bandas.csv.}
  \note{
    \textbf{Qué decir (1 min).} ¿Por qué ninguna de nuestras alternativas pasa de 0,78? Tres pruebas. A: si le damos
    al modelo el ingreso real, el F1 sube a casi 0,9; lo que falta es información sobre el
    ingreso. B: un tercio de los pobres tiene un ingreso a menos de 20\,\% de la línea.
    Para ellos, un hogar pobre y uno no pobre se ven casi iguales en la encuesta, y el F1
    cae a 0,65. Y más datos casi no ayudan: cuadruplicarlos sube el F1 una centésima
    (apéndice).
    Además probamos 16 enfoques más, desde redes neuronales hasta modelos automáticos, y
    todos chocan con el mismo techo.\\[3pt]
    \textbf{Pregunta del profesor: si les damos el ingreso real, ¿por qué no llega a 1?}
    \textbf{Respuesta:} Porque el ingreso de las personas no incluye el arriendo que se
    imputa a los propietarios, que sí cuenta en el ingreso del hogar. Esa parte sigue sin
    verse.
  }
\end{frame}

\section{Implicaciones de política}
%% ============================================================
\begin{frame}{Los pobres que se nos escapan están justo en la línea}
  \vspace{-0.2cm}
  \begin{columns}[c]
    \column{0.60\textwidth}
      {\footnotesize\textbf{Pobres por fuera, según su ingreso}\par}
      \includegraphics[width=\linewidth]{excluidos_por_modelo}
    \column{0.38\textwidth}
      \footnotesize
      \begin{itemize}
        \item \textbf{Subcobertura} (pobres por fuera): 18\,\%; seis de cada diez, cerca
          de la línea.
        \item \textbf{Filtración} (reciben sin necesitarlo): 25\,\%, más cerca de la línea
          que el no pobre típico.
      \end{itemize}
  \end{columns}
  \takeaway{Pesando por la brecha de pobreza, la subcobertura baja de 18\,\% a 11\,\%: los que se escapan son los menos pobres.}
  \fuente{Datos apartados, umbral 0,36. Cerca: ingreso $\geq 0{,}8$ veces la línea; extrema: $< 0{,}5$. Logit, 48: con 48 variables. Mezcla sin el rediseño (95\,\% del peso final). Fuente: views/tablas/nivel7/etapa15\_pesos\_bienestar.csv.}
  \note{
    \textbf{Qué decir (1 min).} Cada error tiene un costo humano, pero no todos cuestan
    lo mismo. El modelo deja por fuera al 18\,\% de los pobres, y seis de cada diez de
    ellos tienen un ingreso de al menos 80\,\% de la línea. El rojo, la pobreza extrema,
    es la parte más pequeña. Del otro lado, uno de cada cuatro atendidos no es pobre,
    pero están mucho más cerca de la línea que el no pobre típico: su distancia media
    a la línea es 0,51, frente a 2,83 del no pobre promedio. Si pesamos a cada pobre
    excluido por su brecha de pobreza, la subcobertura baja de 18 a 11\,\%. Cómo se
    pondera: cada pobre excluido pesa su brecha, $1 - \text{ingreso}/\text{línea}$, y la
    subcobertura ponderada es la suma de esas brechas sobre la de todos los pobres. No
    mostramos la filtración ponderada porque su denominador incluye hogares muy por encima
    de la línea y sale pequeña por construcción.\\[3pt]
    \textbf{Pregunta del profesor: ¿el modelo discrimina a algún grupo?}
    \textbf{Respuesta:} Hay más subcobertura en hogares de una o dos personas (34\,\% y
    28\,\%) y en Manizales y Pereira, porque ahí el ingreso depende de una sola persona.
    La agencia podría usar un umbral más bajo para esos grupos.
  }
\end{frame}

\begin{frame}{El umbral es una decisión de política, no estadística}
  \begin{columns}[c]
    \column{0.54\textwidth}
      \includegraphics[width=\linewidth]{curva_cobertura_slides}
    \column{0.44\textwidth}
      \footnotesize
      El F1 supone que ambos errores cuestan igual.
      \begin{itemize}
        \item Umbral de 0,36 a 0,21: subcobertura de 18\,\% a 10\,\%.
        \item Filtración de 25\,\% a 34\,\%; F1 de 0,78 a 0,76.
      \end{itemize}
  \end{columns}
  \takeaway{Si dejar a un pobre por fuera cuesta más, conviene bajar el umbral.}
  \fuente{Modelo final (n18), datos apartados. Umbrales elegidos en OOF. Punteada: focalización perfecta. Fuente: views/tablas/nivel7/umbral\_n18\_*.csv.}
  \note{
    \textbf{Qué decir (1 min).} El umbral que maximiza el F1 trata igual los dos errores.
    Pero un gobierno puede pensar que dejar por fuera a una familia pobre es peor que
    darle el subsidio a una familia que está apenas encima de la línea. Si es así, puede
    bajar el umbral de 0,36 a 0,21: la subcobertura se reduce casi a la mitad, de 18 a
    10\,\%, a cambio de unos nueve puntos más de filtración, y el F1 baja apenas de 0,78
    a 0,76. Y como los filtrados están relativamente cerca de
    la línea, ese costo es moderado. La curva muestra que atendiendo al 30\,\% de los hogares ya cubrimos al
    93\,\% de los pobres.\\[3pt]
    \textbf{Pregunta del profesor: ¿qué umbral recomiendan?}
    \textbf{Respuesta:} Depende de cuánto más cuesta excluir que filtrar, y esa es una
    decisión normativa. Con costos iguales, el del F1; si excluir cuesta el doble, uno más
    bajo.
  }
\end{frame}

\begin{frame}{Una encuesta corta ahorra plata, pero deja por fuera a más pobres extremos}
  \begin{columns}[c]
    \column{0.54\textwidth}
      \includegraphics[width=\linewidth]{frontera_encuesta}
    \column{0.44\textwidth}
      \footnotesize
      Nuestro modelo necesita el módulo de trabajo de cada persona: largo y caro.
      \begin{itemize}
        \item 8 preguntas: F1 de 0,66.
        \item Pobres por fuera: 26\,\% (mejor modelo: 18\,\%).
        \item De ellos, en pobreza extrema: 16\,\% (11\,\%).
      \end{itemize}
  \end{columns}
  \takeaway{Lo que se ahorra en la encuesta lo pagan los más pobres.}
  \fuente{Logit, datos apartados; preguntas elegidas con un algoritmo genético. Kaggle: 0,66 con XGBoost y 10 preguntas. Fuente: views/tablas/nivel7/frontera\_encuesta\_corta.csv, etapa15\_pesos\_bienestar.csv.}
  \note{
    \textbf{Qué decir (1 min).} ¿Se puede usar esto con una encuesta corta y barata?
    Las 8 preguntas, elegidas con un algoritmo genético, son: ocupados, horas,
    cotizantes, quién quiere trabajar más, arriendo pagado y recibido, y dormitorios.
    Buscamos el mejor grupo de preguntas para cada tamaño de encuesta. Con 8 preguntas
    (cuántos trabajan, cuántas horas, cuántos cotizan, el arriendo y los dormitorios) el
    F1 baja de 0,78 a 0,66. Lo más grave no es el número, es a quién excluye: la encuesta
    corta deja por fuera a más pobres, y más de ellos en pobreza extrema. Ahorrar en la
    encuesta se paga con los hogares más vulnerables.\\[3pt]
    \textbf{Pregunta del profesor: ¿no pueden estimar el ingreso por persona también en
    la encuesta corta?}
    \textbf{Respuesta:} No sin el módulo de trabajo, porque ese modelo se alimenta de la
    ocupación, las horas y la cotización de cada persona. Por eso la encuesta corta se
    evaluó sin él.
  }
\end{frame}

\begin{frame}{Recomendación: preguntar bien por el trabajo de cada persona}
  \small
  \begin{enumerate}
    \item \textbf{Medir el trabajo de cada persona}; basta un LightGBM.
    \item \textbf{Umbral según el costo de excluir.}
    \item \textbf{Poco presupuesto:} encuesta corta y verificar cerca del umbral.
    \item \textbf{Vigilar} hogares pequeños, Manizales y Pereira.
  \end{enumerate}
  \takeaway{Medir bien el ingreso vale más que el algoritmo.}
  \fuente{Atendiendo al 30\,\% de los hogares se cubre al 92,9\,\% de los pobres con el modelo, al 87,1\,\% con el Logit y al 81,5\,\% con la encuesta corta (datos apartados). Fuente: views/tablas/nivel7/etapa15\_curva\_cobertura.csv, etapa15\_errores\_subgrupo.csv.}
  \note{
    \textbf{Qué decir (1 min).} Cerramos con lo que le diríamos a una agencia. Primero, la
    inversión que más rinde es preguntar bien por el trabajo de cada persona; con eso, un
    LightGBM solo basta. Segundo, el umbral es una decisión de política: si excluir a un
    pobre es grave, hay que bajarlo; el F1 no debe decidirlo. Tercero, si solo alcanza para una encuesta corta,
    usarla como filtro amplio y verificar en campo a los hogares cercanos al umbral, que
    es donde el modelo duda. Y cuarto, vigilar a los grupos con más subcobertura: hogares
    de una o dos personas, Manizales y Pereira. El valor está en medir bien el ingreso, no
    en el algoritmo: el límite es de información.\\[3pt]
    \textbf{Pregunta del profesor: ¿se pueden verificar en campo las variables clave?}
    \textbf{Respuesta:} En parte: la vivienda y el arriendo se ven en una visita, pero las
    horas y la ocupación son autorreportadas. Por eso proponemos verificar solo a los
    hogares cerca del umbral, donde un error cambia la decisión.
  }
\end{frame}
